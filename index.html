Apps Script — copy & paste into Google

const SHEET_NAME = "Trades";
const FOLDER_NAME = "ObsidianAnalyst_Screenshots";
const HEADERS = ["id","date","time","instrument",
  "direction","entry","exit","sl","tp","contracts",
  "fees","pnl","ticks","rr","riskpct","result",
  "checklist","psychology","tags","notes","screenshot"];

function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    const action = d.action || "add";
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sh = ss.getSheetByName(SHEET_NAME);
    if (!sh) {
      sh = ss.insertSheet(SHEET_NAME);
      sh.appendRow(HEADERS);
      sh.setFrozenRows(1);
    }
    if (action === "upload_image") {
      const blob = Utilities.newBlob(
        Utilities.base64Decode(d.data),
        d.mimeType, d.filename);
      let folder;
      const fi = DriveApp.getFoldersByName(FOLDER_NAME);
      folder = fi.hasNext() ? fi.next()
        : DriveApp.createFolder(FOLDER_NAME);
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK,
        DriveApp.Permission.VIEW);
      const url = "https://drive.google.com/uc?id="
        + file.getId();
      return ok({ url: url });
    }
    if (action === "add") {
      sh.appendRow(HEADERS.map(h => d[h] ?? ""));
      return ok({ id: d.id });
    }
    if (action === "update") {
      const rows = sh.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(d.id)) {
          HEADERS.forEach((h, j) => {
            if (d[h] !== undefined)
              sh.getRange(i+1,j+1).setValue(d[h]);
          });
          return ok({ updated: d.id });
        }
      }
      return err("id not found");
    }
    if (action === "delete") {
      const rows = sh.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(d.id)) {
          sh.deleteRow(i + 1);
          return ok({ deleted: d.id });
        }
      }
      return err("id not found");
    }
    return err("unknown action");
  } catch(e2) { return err(e2.toString()); }
}

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName(SHEET_NAME);
    if (!sh) return ok({ trades: [] });
    const rows = sh.getDataRange().getValues();
    const hdrs = rows[0];
    const trades = rows.slice(1)
      .filter(r => r[0])
      .map(r => {
        const o = {};
        hdrs.forEach((h,i) => o[h] = r[i]);
        return o;
      });
    return ok({ trades: trades.reverse() });
  } catch(e2) { return err(e2.toString()); }
}

function ok(d){return out(Object.assign({status:"ok"},d));}
function err(m){return out({status:"error",msg:m});}
function out(d){
  return ContentService
    .createTextOutput(JSON.stringify(d))
    .setMimeType(ContentService.MimeType.JSON);
}
